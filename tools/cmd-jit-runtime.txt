:el_drive
rem CHUNKED driver (mirror of cmd-interp-runtime's el_drive, minus the interp executor): a cmd
rem backward goto costs a FULL-FILE scan, so the dispatch loop is a fixed `for /l` whose body is
rem flag-guarded, drained via ELDONE -- one backward goto per 1000 dispatches. This was the last
rem unchunked driver (load-cmd/eval-cmd only -- the production engines use the interp runtime's);
rem it made the cmd-jit test legs ~2.5x slower than the same compile under the chunked driver.
set "ELDONE="
:eld_chunk
for /l %%Z in (1,1,1000) do if not defined ELDONE (
  set "ACTION="
  call "!CURFN!_pc!PC!.cmd"
  if "!ACTION!"=="call" (
    set "RSF!RSP!=!CURFN!" & set "RSC!RSP!=!RPC!" & set "RSB!RSP!=!FP!" & set "RSL!RSP!=!CLO!"
    set /a RSP+=1
    set "FP=!NFP!" & set "PC=0" & set "CLO="
    if "!CALLEE:~0,2!"=="K:" (call :eld_clo) else if "!CALLEE:~0,2!"=="C:" (set "CURFN=!CALLEE:~2!") else set "CURFN=!CALLEE!"
  ) else if "!ACTION!"=="ret" (
    if !RSP!==0 (set "CURFN=HALT" & set "ELDONE=1") else (
      set /a RSP-=1
      call set "FP=%%RSB!RSP!%%" & call set "CURFN=%%RSF!RSP!%%" & call set "PC=%%RSC!RSP!%%" & call set "CLO=%%RSL!RSP!%%"
    )
  ) else if "!ACTION!"=="apply" (
    set "RSF!RSP!=!CURFN!" & set "RSC!RSP!=!RPC!" & set "RSB!RSP!=!FP!" & set "RSL!RSP!=!CLO!"
    set /a RSP+=1
    set "FP=!NFP!" & set "PC=0" & set "CLO="
    call :eld_apply_spread
    if "!CALLEE:~0,2!"=="K:" (call :eld_clo) else if "!CALLEE:~0,2!"=="C:" (set "CURFN=!CALLEE:~2!") else set "CURFN=!CALLEE!"
  ) else if "!ACTION!"=="" call :eld_noaction
  if "!CURFN!"=="HALT" set "ELDONE=1"
)
if not defined ELDONE goto eld_chunk
set "ELDONE="
goto :eof
:eld_noaction
echo drive: !CURFN! pc !PC! yielded no ACTION ^(unbound global / missing fn?^) 1>&2
set "CURFN=HALT"
goto :eof
:eld_apply_spread
rem spread APLIST into the callee frame; publish the count (variadic callees read ARGC)
set "_ai=0" & set "_ac=!APLIST!"
:eld_ap_loop
if "!_ac!"=="NIL" (set "ARGC=!_ai!" & goto :eof)
call :hp_car "!_ac!"
set /a _i=!FP!+!_ai!
set "F!_i!=!R!"
call :hp_cdr "!_ac!"
set "_ac=!R!"
set /a _ai+=1
goto eld_ap_loop
:eld_clo
set "CLO=!CALLEE:~2!"
call :hp_car "P:!CLO!"
set "CURFN=!R:~2!"
goto :eof

rem ---- value renderer: byte-identical to eval-sh's _relem/_rlist (sets ELR) ----------------
:el_relem
rem %1 = recursion depth (for unique scratch names), %2 = tagged value -> ELR
set "ev=%~2"
if "!ev!"=="NIL" set "ELR=()" & goto :eof
set "ep=!ev:~0,2!"
if "!ep!"=="I:" set "ELR=!ev:~2!" & goto :eof
if "!ep!"=="S:" set "ELR=!ev:~2!" & goto :eof
if "!ep!"=="T:" set "ELR=!ev:~2!" & goto :eof
if "!ep!"=="K:" set "ELR=<closure>" & goto :eof
if "!ep!"=="C:" set "ELR=<fn:!ev:~2!>" & goto :eof
if "!ep!"=="P:" goto el_relem_pair
set "ELR=!ev!" & goto :eof
:el_relem_pair
set /a ND=%1+1 & call :el_rlist !ND! "!ev!"
set "ELR=(!ELR!)"
goto :eof

:el_rlist
rem %1 = depth, %2 = P: ref -> ELR = "e e e" (proper) or "e e . t" (improper)
set "_r%1_lst=%~2"
set "_r%1_acc="
set "_r%1_first=1"
:el_rl2
call :hp_car "!_r%1_lst!"
set /a ND=%1+1 & call :el_relem !ND! "!R!"
if "!_r%1_first!"=="1" (set "_r%1_acc=!ELR!") else (set "_r%1_acc=!_r%1_acc! !ELR!")
set "_r%1_first=0"
call :hp_cdr "!_r%1_lst!"
set "_r%1_tl=!R!"
if "!_r%1_tl!"=="NIL" set "ELR=!_r%1_acc!" & goto :eof
if "!_r%1_tl:~0,2!"=="P:" set "_r%1_lst=!_r%1_tl!" & goto el_rl2
set /a ND=%1+1 & call :el_relem !ND! "!_r%1_tl!"
set "ELR=!_r%1_acc! . !ELR!"
goto :eof
